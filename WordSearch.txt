class Solution {
    public boolean exist(char[][] board, String word) {
        
        int[][] dirs = {{-1,0},{1,0},{0,-1},{0,1}};
        for(int i =0; i < board.length ; i++){
            for(int j=0 ; j< board[0].length ; j++){
                if(board[i][j] == word.charAt(0)){
                    if(helper(board,i,j ,0, word, dirs))return true;
                }
            }
        }

        return false;
    }

    private boolean helper(char[][] board, int i, int j, int idx, String word, int[][] dirs){

        //base
        if(idx == word.length()) return true;

        //logic
        char curr = board[i][j];
        if(curr == word.charAt(idx)){
            if(word.length() == 1) return true;
            // action 
            board[i][j] = '#';
            // recurse 
            for(int[] dir : dirs){
                int r = i + dir[0];
                int c = j + dir[1];
                if(r >= 0 && r < board.length && c >=0 && c < board[0].length){
                    if(helper(board,r,c,idx+1,word,dirs)) return true;
                }
            }
            // backtrack
            board[i][j] = word.charAt(idx);
        }
        return false;
    }
}
