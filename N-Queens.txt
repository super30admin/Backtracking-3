class Solution {
    List<List<String>> result;
    public List<List<String>> solveNQueens(int n) {
        this.result = new ArrayList<>();
        boolean[][] board = new boolean[n][n];

            helper(board, 0,n);
        return result;
    }

    private void helper(boolean[][] board, int idx,int n){
        //base
        if(idx == board.length){
            List<String> row = new ArrayList<>();
            for(int i=0; i< n; i++){
                StringBuilder sb = new StringBuilder();
                for(int j=0; j< n; j++){
                    if(board[i][j] == true) sb.append("Q");
                    else sb.append(".");
                }
                row.add(sb.toString());
            }
            result.add(row);
            return ;
        }
        //logic
        for(int j=0; j< n ;j++){

            if(isValid(board,idx,j,n)){
                //action
                board[idx][j] = true;
                //recurse 
                helper(board,idx+1,n);
                //backtrack
                board[idx][j] = false;
            }
        }
    }
     private boolean isValid(boolean[][] board, int i, int j, int n){
        
        int row = i;
        int col = j; 

        while(row >= 0){
            if(board[row][col] == true) return false;
            row --;
        }
        row = i;
        col = j;
        while(row >= 0 && col >= 0){
            if(board[row][col] == true) return false;
            row--;
            col--;
        }
        row = i;
        col = j;
        while(row >= 0 && col < n){
            if(board[row][col] == true) return false;
            row--;
            col++;
        }
        return true;
    }
}
